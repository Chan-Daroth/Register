function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data  = JSON.parse(e.parameter.data);

    if (sheet.getLastRow() === 0) {
      var h = ['Timestamp','Full Name','National ID','ID Expiry',
               'Phone','Telegram','Commune / Address',
               'Preferred Service','Interview Date','Time Slot',
               'Confirmed Attendance','Status'];
      sheet.appendRow(h);
      sheet.getRange(1,1,1,h.length)
        .setFontWeight('bold')
        .setBackground('#00b14f')
        .setFontColor('#ffffff');
    }

    sheet.appendRow([
      new Date().toLocaleString('en-GB'),
      data.fullName, data.nationalId, data.idExpiry,
      data.phone,    data.telegram,   data.commune,
      data.service,  data.days,       data.timeSlot,
      data.confirmed,'New'
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({status:'success'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error',msg:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet  = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data   = sheet.getDataRange().getValues();
    var counts = {};
    if (data.length > 1) {
      var headers = data[0].map(function(h){ return String(h).trim(); });
      var dIdx = headers.indexOf('Interview Date');
      if (dIdx < 0) dIdx = headers.indexOf('Interview Day(s)');
      var tIdx = headers.indexOf('Time Slot');
      if (dIdx >= 0 && tIdx >= 0) {
        for (var i = 1; i < data.length; i++) {
          var day  = String(data[i][dIdx]).trim();
          var time = String(data[i][tIdx]).trim();
          if (!day || !time) continue;
          var sess = time.toLowerCase().indexOf('pm') >= 0 ? 'PM' : 'AM';
          var key  = day + '|' + sess;
          counts[key] = (counts[key] || 0) + 1;
        }
      }
    }
    return ContentService
      .createTextOutput(JSON.stringify({status:'alive',counts:counts}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'alive',counts:{}}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
